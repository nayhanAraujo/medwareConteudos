FLUXO
1) Cadastro/recadastro na tela frmScriptLaudo.SalvarModeloLaudo()
2) A DLL selecionada é lida do disco e convertida em Base64
3) O Base64 é atribuído a mScriptLaudo.EstruturaScript
4) SalvarComEstrutura() chama ScriptLaudo_Salvar() (SqlBd)
5) UPDATE OR INSERT em SCRIPTLAUDO, gravando o Base64 no campo ESTRUTURASCRIPT
   (via parâmetro @ESTRUTURASCRIPT)

public static string FileInString64(string caminhoArquivo)
{
    string retorno = string.Empty;
    if (!string.IsNullOrEmpty(caminhoArquivo) && ArquivoExiste(caminhoArquivo))
        retorno = Convert.ToBase64String(File.ReadAllBytes(caminhoArquivo));
    return retorno;
}

public static MemoryStream String64InStream(string dados)
{
    byte[] imageBytes = Convert.FromBase64String(dados);
    MemoryStream ms = new MemoryStream(imageBytes, 0, imageBytes.Length);
    ms.Write(imageBytes, 0, imageBytes.Length);
    return ms;
}

4) clsScriptLaudo.ScriptLaudo_Salvar (Medware.Clinicas.SqlBd) - grava ESTRUTURASCRIPT

public int ScriptLaudo_Salvar(int codScriptLaudo, int liberado, string titulo, string scriptLaudo,
    string estruturaScript, bool flagLaudo, bool flagTextoLaudo, bool flagAutoGeracaoLaudo,
    bool flagCapturaImagem, bool flagCapturaTelaLaudo, bool flagVisualizacao, bool flagAssociacao,
    bool flagAutoAssociacao, bool flagAdicaoArquivo, bool flagMontaLaudo, bool flagCapturaVideo,
    bool flagEcoStress, int totFotos, bool ativo, bool impressaoAutomatica, int codUsuario, int tipoScript)
{
    StringBuilder sql = new StringBuilder();
    List<FbParameter> parametros = new List<FbParameter>();
    MdwBdInst conexao = new MdwBdInst();

    parametros.Add(new FbParameter("@ESTRUTURASCRIPT", estruturaScript));

    sql.AppendLine("UPDATE OR INSERT INTO SCRIPTLAUDO(");
    sql.AppendLine("  CODSCRIPTLAUDO,");
    sql.AppendLine("  LIBERADO,");
    sql.AppendLine("  TITULO,");
    sql.AppendLine("  SCRIPTLAUDO,");
    sql.AppendLine("  ESTRUTURASCRIPT,");
    sql.AppendLine("  ATIVO,");
    sql.AppendLine("  FLAGLAUDO,");
    sql.AppendLine("  FLAGTEXTOLAUDO,");
    sql.AppendLine("  FLAGAUTOGERACAOLAUDO,");
    sql.AppendLine("  FLAGCAPTURAIMAGEM,");
    sql.AppendLine("  FLAGCAPTURATELALAUDO,");
    sql.AppendLine("  FLAGVISUALIZACAO,");
    sql.AppendLine("  FLAGASSOCIACAO,");
    sql.AppendLine("  FLAGAUTOASSOCIACAO,");
    sql.AppendLine("  FLAGADICAOARQUIVO,");
    sql.AppendLine("  FLAGMONTALAUDO,");
    sql.AppendLine("  FLAGCAPTURAVIDEO,");
    sql.AppendLine("  FLAGECOSTRESS,");
    sql.AppendLine("  TOTFOTOS,");
    sql.AppendLine("  TIPOSCRIPT,");
    sql.AppendLine("  IMPRESSAOAUTOMATICA,");
    sql.AppendLine("  CODUSUARIO,");
    sql.AppendLine("  DTHRULTMODIFICACAO)");
    sql.AppendLine("VALUES(");
    sql.AppendFormat("{0},", codScriptLaudo.Equals(0) ? "Null" : codScriptLaudo.ToString()).AppendLine();
    sql.AppendFormat("{0},", liberado.ToString()).AppendLine();
    sql.AppendFormat("{0},", titulo.StringBD()).AppendLine();
    sql.AppendFormat("{0},", scriptLaudo.StringBD()).AppendLine();
    sql.AppendLine("@ESTRUTURASCRIPT,");
    sql.AppendFormat("{0},", ativo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagLaudo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagTextoLaudo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagAutoGeracaoLaudo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagCapturaImagem ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagCapturaTelaLaudo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagVisualizacao ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagAssociacao ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagAutoAssociacao ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagAdicaoArquivo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagMontaLaudo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagCapturaVideo ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", flagEcoStress ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", totFotos.ToString()).AppendLine();
    sql.AppendFormat("{0},", tipoScript.ToString()).AppendLine();
    sql.AppendFormat("{0},", impressaoAutomatica ? "-1" : "0").AppendLine();
    sql.AppendFormat("{0},", codUsuario.ToString()).AppendLine();
    sql.AppendLine("  CURRENT_TIMESTAMP)").AppendLine();
    sql.AppendLine("Returning CODSCRIPTLAUDO;");

    int retorno = Convert.ToInt32(conexao.Executar(sql.ToString(), parametros, "CODSCRIPTLAUDO", FbDbType.Integer));
    return retorno;
}

Observações:
- Firebird usa boolean como -1 (true) / 0 (false).
- ESTRUTURASCRIPT sempre via parâmetro (@ESTRUTURASCRIPT) por ser BLOB/texto grande.
- TIPOSCRIPT: 1 = Texto/HTML, 2 = DLL .NET, 3 = JSON.
